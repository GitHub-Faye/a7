### Markdown生成代码修改建议
```python
def generate_markdown_with_visual_hierarchy(knowledge_point, level=1):
    """生成包含视觉层级标识的Markdown内容"""
    # 添加CSS类标记当前层级
    markdown = f"# {knowledge_point.title} {{.level-{level}}}\n\n"
    
    # 添加带有缩进的内容
    content_lines = knowledge_point.content.split('\n')
    indented_content = '\n'.join([line for line in content_lines])
    markdown += f"{indented_content}\n\n"
    
    # 处理子知识点，增加层级
    for child in knowledge_point.children.all().order_by('-importance'):
        # 添加明确的分隔符和层级标记
        markdown += f"\n---\n<!-- level: {level+1} -->\n\n"
        # 递归处理子知识点
        markdown += generate_markdown_with_visual_hierarchy(child, level+1)
    
    return markdown
```
